--Sesion 15.2
/*Crea una tabla Movimientos particionada por hash usando la columna CodigoCliente (4 particiones). 
Inserta datos de Pedidos y escribe una consulta que muestre el total de ventas por cliente, 
verificando que las particiones se usen.*/

DROP TABLE Movimientos;

CREATE TABLE Movimientos (
    MovimientoID NUMBER PRIMARY KEY,
    CodigoCliente NUMBER,
    Importe NUMBER,
    FechaRegistro DATE
)
PARTITION BY HASH (CodigoCliente) 
PARTITIONS 4;

INSERT INTO Movimientos (MovimientoID, CodigoCliente, Importe)
SELECT 
    PedidoID,
    ClienteID,
    Total
FROM Pedidos
WHERE ClienteID IS NOT NULL;

-- Consulta que muestra el total de ventas por cliente
SELECT CodigoCliente, SUM(Importe) as ImporteTotal
FROM Movimientos
GROUP BY CodigoCliente;

-- PASO 1: Conectar como sysdba
-- sqlplus sys/oracle@//localhost:1521/XEPDB1 as sysdba

-- Limpiar objetos previos si ya existen
DROP USER usr_analista_datos CASCADE;
DROP ROLE rol_consultas_ventas;

-- Crear usuario
CREATE USER usr_analista_datos IDENTIFIED BY datos123;
GRANT CONNECT TO usr_analista_datos;

-- Crear rol y asignar permisos
CREATE ROLE rol_consultas_ventas;
GRANT SELECT ON curso_topicos.Clientes TO rol_consultas_ventas;
GRANT SELECT ON curso_topicos.Pedidos TO rol_consultas_ventas;
GRANT SELECT ON curso_topicos.Productos TO rol_consultas_ventas;
GRANT SELECT ON curso_topicos.DetallesPedidos TO rol_consultas_ventas;
GRANT INSERT ON curso_topicos.Pedidos TO rol_consultas_ventas;

-- Asignar rol al usuario
GRANT rol_consultas_ventas TO usr_analista_datos;

-- PASO 2: Abrir nueva sesion como usr_analista_datos

/*Configura auditoría para monitorear las acciones de usr_analista_datos al consultar la tabla 
Clientes y al insertar en la tabla Pedidos. Realiza algunas acciones y verifica los 
registros de auditoría.*/

-- PASO 1: Conectar como sysdba
-- sqlplus sys/oracle@//localhost:1521/XEPDB1 as sysdba

-- Crear politica de auditoria
CREATE AUDIT POLICY pol_auditoria_usuario_analista
    ACTIONS SELECT ON curso_topicos.Clientes,
             INSERT ON curso_topicos.Pedidos;

-- Habilitar la politica para usr_analista_datos
AUDIT POLICY pol_auditoria_usuario_analista BY usr_analista_datos;

-- PASO 2: Conectar como usr_analista_datos y realizar acciones
-- sqlplus usr_analista_datos/datos123@//localhost:1521/XEPDB1

SELECT * FROM curso_topicos.Clientes;
INSERT INTO curso_topicos.Pedidos (PedidoID, ClienteID, Total, FechaPedido)
VALUES (111, 2, 2000, TO_DATE('2025-06-15', 'YYYY-MM-DD'));

-- PASO 3: Verificar registros de auditoria como sysdba
-- sqlplus sys/oracle@//localhost:1521/XEPDB1 as sysdba

SELECT event_timestamp, dbusername, action_name, object_name, return_code
FROM unified_audit_trail
WHERE dbusername = 'USR_ANALISTA_DATOS'
ORDER BY event_timestamp DESC;

SET SERVEROUTPUT ON;

CREATE OR REPLACE FUNCTION calcular_edad_cliente(
    v_id_cliente IN NUMBER
) RETURN NUMBER AS
    fec_nacimiento DATE;
    num_edad NUMBER;
BEGIN
    -- Obtenemos la fecha de nacimiento del registro solicitado
    SELECT FechaNacimiento 
    INTO fec_nacimiento 
    FROM Clientes 
    WHERE ClienteID = v_id_cliente;
    
    -- Validamos en caso de que exista el registro, pero el campo esté vacío
    IF fec_nacimiento IS NULL THEN
        RAISE_APPLICATION_ERROR(-20001, 'El registro del ClienteID ' || v_id_cliente || ' existe, pero no posee fecha de nacimiento.');
    END IF;
    
    -- Calculamos los meses entre la fecha actual y la fecha de nacimiento, para luego truncarlo a años
    num_edad := TRUNC(MONTHS_BETWEEN(SYSDATE, fec_nacimiento) / 12);
    RETURN num_edad;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No existen coincidencias para el ClienteID: ' || v_id_cliente);
        RETURN NULL;
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Ocurrió un fallo general: ' || SQLERRM);
        RETURN NULL;
END;
/

-- Bloque de prueba de la función
DECLARE
    edad_resultado NUMBER; 
BEGIN
    edad_resultado := calcular_edad_cliente(1);
    
    IF edad_resultado IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('La edad calculada para el cliente es: ' || edad_resultado || ' años.');
    END IF;
END;
/

SET SERVEROUTPUT ON;

CREATE OR REPLACE FUNCTION obtener_precio_promedio RETURN NUMBER AS
    num_promedio NUMBER;
BEGIN
    -- Se asigna a la variable el promedio calculado de los productos
    SELECT AVG(Precio) 
    INTO num_promedio 
    FROM Productos;
    
    RETURN num_promedio;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error de procesamiento: ' || SQLERRM);
        RETURN NULL;
END;
/

-- Consulta de prueba aplicando la función creada
SELECT 
    ProductoID, 
    Nombre, 
    Precio
FROM Productos
WHERE Precio > obtener_precio_promedio();
